26  ! Returns saturation specific humidity for given temperature and pressure (at some atmospheric level or at surface)
27  ! Uses Eq. 2.27 of Fundamentals of Atmospheric Modeling for the saturated case, but the saturation vapor pressure

29  !

36  REAL(wp), INTENT(in) :: temperature ! Air temperature [K]
37  REAL(wp), INTENT(in) :: pressure ! Pressure [Pa]
38 
39  INTEGER :: it
40  REAL(wp) :: tluc
41 
42  it = nint(temperature*1000._wp)
43 !!$ it = MIN(MAX(NINT(temperature*1000._wp), 50000),400000) ! Temporary fix
44  tluc = tlucua(it)
45 
46  IF (it >= jptlucu1 .AND. it <= jptlucu2) THEN
47  qsat = tluc / (pressure - vtmpc1*tluc)
48  ELSE
49  qsat = huge(1._wp)
50  END IF
51 
52  END FUNCTION qsat
